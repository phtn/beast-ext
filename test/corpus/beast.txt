  empty
    EmptyState(label={fallback})
  each option in " key ".split(" ") key option
    span #{option}
  switch status
    case "ready"
      ReadyView
    default
      UnknownView
  try
    AsyncView(resource={resource})
  pending
    p Loading
  catch (error, reset)
    button(onClick={reset}) #{String(error)}
